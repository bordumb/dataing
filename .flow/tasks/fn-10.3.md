```python
import asyncio
import signal
from uuid import uuid4
from arq import Worker

from dataing.core.queue import get_redis_settings
from dataing.adapters.db.app_db import AppDatabase

WORKER_ID = str(uuid4())
shutdown_event = asyncio.Event()

async def run_investigation(ctx: dict, investigation_id: str, tenant_id: str) -> dict:
    """Main job handler - executes investigation workflow."""
    # Implementation in later tasks
    pass

async def startup(ctx: dict):
    """Initialize worker resources."""
    ctx["db"] = await AppDatabase.create()

    # Register signal handlers
    loop = asyncio.get_event_loop()
    for sig in (signal.SIGTERM, signal.SIGINT):
        loop.add_signal_handler(sig, lambda: shutdown_event.set())

async def shutdown(ctx: dict):
    """Cleanup worker resources."""
    if "db" in ctx:
        await ctx["db"].close()

class WorkerSettings:
    functions = [run_investigation]
    queue_name = "investigations"
    redis_settings = get_redis_settings()

    on_startup = startup
    on_shutdown = shutdown

    max_jobs = 5
    job_timeout = 7200  # 2 hours
    job_completion_wait = 30  # Align with K8s terminationGracePeriodSeconds
    health_check_interval = 30
    handle_signals = True
    retry_jobs = False  # We handle retries via job status

if __name__ == "__main__":
    Worker(WorkerSettings).run()
```

### Notes

- `job_completion_wait=30` ensures in-flight jobs complete before shutdown
- `retry_jobs=False` because we use `investigation_jobs.attempts` for retry control
- Worker ID is UUID for uniqueness across pods

# fn-47.6 Add CLI commands: schedule create, schedule list

## Description

Add CLI commands `dataing schedule create` and `dataing schedule list` so users can manage scheduled investigations from the terminal. The CLI calls the schedule API endpoints created in fn-47.4. This enables headless automation and CI/CD integration for setting up investigation schedules.

Depends on: fn-47.4 (API routes must exist for the CLI to call).

### Implementation

1. Create `python-packages/dataing/src/dataing/entrypoints/cli/schedule.py` with two commands:

```python
"""CLI commands for managing scheduled investigations."""

import httpx
import json
import sys

def schedule_create(
    asset_id: str,
    goal: str,
    cron: str,
    timezone: str = "UTC",
    notify_always: bool = False,
    channels: str | None = None,  # JSON string
    api_url: str = "http://localhost:8000",
    api_key: str | None = None,
) -> None:
    """Create a new scheduled investigation."""
    payload = {
        "asset_id": asset_id,
        "goal": goal,
        "cron_expression": cron,
        "timezone": timezone,
        "notify_always": notify_always,
        "notification_channels": json.loads(channels) if channels else [],
    }
    resp = httpx.post(
        f"{api_url}/api/schedules",
        json=payload,
        headers=_auth_headers(api_key),
    )
    resp.raise_for_status()
    data = resp.json()
    print(f"Created schedule {data['id']}")
    print(f"  Next run: {data.get('next_run_at', 'N/A')}")

def schedule_list(
    active_only: bool = False,
    format: str = "table",
    api_url: str = "http://localhost:8000",
    api_key: str | None = None,
) -> None:
    """List all scheduled investigations."""
    params = {"active_only": "true"} if active_only else {}
    resp = httpx.get(
        f"{api_url}/api/schedules",
        params=params,
        headers=_auth_headers(api_key),
    )
    resp.raise_for_status()
    schedules = resp.json()
    if format == "json":
        print(json.dumps(schedules, indent=2))
        return
    for s in schedules:
        status = "active" if s["is_active"] else "paused"
        print(
            f"{s['id'][:8]}  {status:<8}  {s['cron_expression']:<15}  "
            f"{s['goal'][:40]}"
        )
```

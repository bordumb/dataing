**Deprecation approach:**
```python
import warnings

def __getattr__(name: str):
    deprecated = {
        "BranchingSignalHandler": "Use Engine instead",
        "DefaultSignalHandler": "Use Engine instead",
        "SignalHandler": "Use Engine instead",
        "SignalResult": "Use events instead",
    }
    if name in deprecated:
        warnings.warn(
            f"{name} is deprecated. {deprecated[name]}",
            DeprecationWarning,
            stacklevel=2
        )
        # Still return the class for backward compat
        from maestro import handlers
        return getattr(handlers, name)
    raise AttributeError(f"module 'maestro' has no attribute '{name}'")
```

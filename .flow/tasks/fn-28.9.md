```python
"""Error handling for CLI."""
from __future__ import annotations

import typer
from rich.console import Console

from dataing_sdk.exceptions import (
    DataingError,
    AuthError,
    NotFoundError,
    RateLimitError,
    ValidationError,
    AmbiguousAssetError,
    ConnectionError,
)

console = Console()


class CLIError(Exception):
    """Base CLI error with exit code."""
    def __init__(self, message: str, exit_code: int = 1):
        self.message = message
        self.exit_code = exit_code
        super().__init__(message)


def handle_sdk_error(exc: DataingError) -> None:
    """Map SDK exceptions to CLI output and exit."""
    if isinstance(exc, AuthError):
        console.print("[red]Authentication failed.[/red] Check your API key.")
        console.print("[dim]Hint: Run 'dataing init' to configure credentials.[/dim]")
        raise typer.Exit(1)

    if isinstance(exc, NotFoundError):
        console.print(f"[red]Not found:[/red] {exc.message}")
        raise typer.Exit(1)

    if isinstance(exc, RateLimitError):
        retry_after = exc.retry_after or 60
        console.print(f"[yellow]Rate limited.[/yellow] Retry after {retry_after} seconds.")
        raise typer.Exit(1)

    if isinstance(exc, ValidationError):
        console.print(f"[red]Invalid input:[/red] {exc.message}")
        if exc.details:
            for field, errors in exc.details.items():
                console.print(f"  • {field}: {', '.join(errors)}")
        raise typer.Exit(2)  # Exit 2 for usage errors

    if isinstance(exc, AmbiguousAssetError):
        console.print(f"[yellow]Ambiguous asset.[/yellow] Multiple matches found:")
        for candidate in exc.candidates:
            console.print(f"  • {candidate}")
        console.print("\n[dim]Hint: Specify the full asset name or datasource.[/dim]")
        raise typer.Exit(1)

    if isinstance(exc, ConnectionError):
        console.print(f"[red]Connection failed:[/red] {exc.message}")
        console.print("[dim]Hint: Check network connectivity and backend URL.[/dim]")
        raise typer.Exit(1)

    # Generic error
    console.print(f"[red]Error:[/red] {exc.message}")
    raise typer.Exit(1)


def cli_error_handler(func):
    """Decorator to wrap commands with error handling."""
    import functools

    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except DataingError as e:
            handle_sdk_error(e)
        except CLIError as e:
            console.print(f"[red]Error:[/red] {e.message}")
            raise typer.Exit(e.exit_code)
        except KeyboardInterrupt:
            console.print("\n[dim]Interrupted.[/dim]")
            raise typer.Exit(130)  # Standard exit code for Ctrl+C

    return wrapper
```

### Exit Code Convention

- `0` - Success
- `1` - General error (auth, not found, connection, etc.)
- `2` - Usage/validation error (bad input)
- `130` - Interrupted (Ctrl+C)

### Usage

Apply decorator to commands or wrap in try/except:

```python
@app.command()
@cli_error_handler
def my_command(...):
    ...
```

```python
def send_message(self, investigation_id: str, message: str) -> dict[str, Any]:
    """Send a user message to an investigation.

    Args:
        investigation_id: The investigation UUID
        message: The user message text

    Returns:
        Response with status and investigation_id
    """
    response = self._request(
        "POST",
        f"/api/v1/investigations/{investigation_id}/messages",
        json={"message": message},
    )
    return response.json()
```

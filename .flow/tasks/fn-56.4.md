Add dependency for SelfDebugAgent:
```python
from dataing.agents.self_debug import SelfDebugAgent

_self_debug_agent: SelfDebugAgent | None = None

def get_self_debug_agent() -> SelfDebugAgent:
    global _self_debug_agent
    if _self_debug_agent is None:
        github_token = os.getenv("GITHUB_TOKEN")
        repo_path = os.getenv("DATAING_REPO_PATH", ".")
        _self_debug_agent = SelfDebugAgent(
            github_token=github_token,
            repo_path=repo_path,
        )
    return _self_debug_agent
```

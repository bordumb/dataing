# fn-48.2 Create SQLAlchemy model and threshold repository

## Description

Create SQLAlchemy ORM models for the `thresholds` and `threshold_evaluations` tables, plus a `ThresholdRepository` class that encapsulates all database operations. The repository provides the domain interface used by the evaluation engine, API routes, and CLI.

Depends on: fn-48.1 (migration must exist first).

### Implementation

1. Create `python-packages/dataing/src/dataing/models/threshold.py`:

```python
"""Threshold persistence models."""

from datetime import datetime
from decimal import Decimal
from typing import TYPE_CHECKING
from uuid import UUID

from sqlalchemy import Boolean, ForeignKey, Integer, Numeric, String, Text
from sqlalchemy.orm import Mapped, mapped_column, relationship

from dataing.models.base import BaseModel

if TYPE_CHECKING:
    from dataing.models.tenant import Tenant
    from dataing.models.user import User


class Threshold(BaseModel):
    """Persisted threshold configuration."""

    __tablename__ = "thresholds"

    tenant_id: Mapped[UUID] = mapped_column(ForeignKey("tenants.id"), nullable=False)
    asset_id: Mapped[UUID] = mapped_column(nullable=False)
    metric_name: Mapped[str] = mapped_column(String(255), nullable=False)
    metric_source: Mapped[str] = mapped_column(String(50), default="query")
    metric_query: Mapped[str | None] = mapped_column(Text, nullable=True)
    operator: Mapped[str] = mapped_column(String(10), nullable=False)
    value: Mapped[Decimal] = mapped_column(Numeric, nullable=False)
    severity: Mapped[str] = mapped_column(String(20), default="warning")
    cooldown_hours: Mapped[int] = mapped_column(Integer, default=24)
    hysteresis_count: Mapped[int] = mapped_column(Integer, default=1)
    evaluation_cron: Mapped[str] = mapped_column(String(100), default="0 * * * *")
    is_active: Mapped[bool] = mapped_column(Boolean, default=True)
    state: Mapped[str] = mapped_column(String(20), default="normal")
    consecutive_breaches: Mapped[int] = mapped_column(Integer, default=0)
    last_evaluated_at: Mapped[datetime | None] = mapped_column(nullable=True)
    last_triggered_at: Mapped[datetime | None] = mapped_column(nullable=True)
    last_value: Mapped[Decimal | None] = mapped_column(Numeric, nullable=True)
    created_by: Mapped[UUID | None] = mapped_column(
        ForeignKey("users.id"), nullable=True
    )

    # Relationships
    tenant: Mapped["Tenant"] = relationship("Tenant")
    created_by_user: Mapped["User | None"] = relationship("User")
    evaluations: Mapped[list["ThresholdEvaluation"]] = relationship(
        back_populates="threshold", cascade="all, delete-orphan"
    )


class ThresholdEvaluation(BaseModel):
    """Record of a single threshold evaluation."""

    __tablename__ = "threshold_evaluations"

    threshold_id: Mapped[UUID] = mapped_column(
        ForeignKey("thresholds.id", ondelete="CASCADE"), nullable=False
    )
    tenant_id: Mapped[UUID] = mapped_column(nullable=False)
    measured_value: Mapped[Decimal] = mapped_column(Numeric, nullable=False)
    is_breach: Mapped[bool] = mapped_column(Boolean, nullable=False)
    previous_state: Mapped[str] = mapped_column(String(20), nullable=False)
    new_state: Mapped[str] = mapped_column(String(20), nullable=False)
    investigation_id: Mapped[UUID | None] = mapped_column(nullable=True)

    # Relationships
    threshold: Mapped["Threshold"] = relationship(back_populates="evaluations")
```

2. Create `python-packages/dataing/src/dataing/core/threshold/__init__.py` and `repository.py` with `ThresholdRepository`:
   - `create(tenant_id, asset_id, metric_name, operator, value, ...) -> Threshold`
   - `get(threshold_id, tenant_id) -> Threshold | None`
   - `list_by_tenant(tenant_id, active_only=False, asset_id=None) -> list[Threshold]`
   - `update(threshold_id, tenant_id, **fields) -> Threshold`
   - `delete(threshold_id, tenant_id) -> bool`
   - `get_active_thresholds_for_evaluation() -> list[Threshold]` -- all active thresholds across tenants
   - `record_evaluation(threshold_id, measured_value, is_breach, prev_state, new_state, investigation_id=None) -> ThresholdEvaluation`
   - `update_state(threshold_id, state, consecutive_breaches, last_value, last_triggered_at=None) -> None`
   - `get_evaluation_history(threshold_id, limit=50) -> list[ThresholdEvaluation]`

3. Register the models in `python-packages/dataing/src/dataing/models/__init__.py`.

# fn-48.7 Add CLI commands: threshold create, threshold list

## Description

Add CLI commands `dataing threshold create` and `dataing threshold list` so users can manage thresholds from the terminal. The CLI calls the threshold API endpoints created in fn-48.3. This enables scripted setup of monitoring thresholds and integration with infrastructure-as-code workflows.

Depends on: fn-48.3 (API routes must exist for the CLI to call).

### Implementation

1. Create `python-packages/dataing/src/dataing/entrypoints/cli/threshold.py` with two commands:

```python
"""CLI commands for managing thresholds."""

import httpx
import json
import sys

def threshold_create(
    asset_id: str,
    metric_name: str,
    operator: str,
    value: float,
    severity: str = "warning",
    cooldown_hours: int = 24,
    hysteresis_count: int = 1,
    evaluation_cron: str = "0 * * * *",
    metric_query: str | None = None,
    api_url: str = "http://localhost:8000",
    api_key: str | None = None,
) -> None:
    """Create a new threshold.

    Example:
        dataing threshold create \\
          --asset-id "..." \\
          --metric-name null_rate \\
          --operator gt \\
          --value 0.05 \\
          --severity critical \\
          --cooldown-hours 12 \\
          --hysteresis-count 3
    """
    payload = {
        "asset_id": asset_id,
        "metric_name": metric_name,
        "operator": operator,
        "value": value,
        "severity": severity,
        "cooldown_hours": cooldown_hours,
        "hysteresis_count": hysteresis_count,
        "evaluation_cron": evaluation_cron,
    }
    if metric_query:
        payload["metric_query"] = metric_query
    resp = httpx.post(
        f"{api_url}/api/thresholds",
        json=payload,
        headers=_auth_headers(api_key),
    )
    resp.raise_for_status()
    data = resp.json()
    print(f"Created threshold {data['id']}")
    print(f"  Metric: {data['metric_name']} {data['operator']} {data['value']}")
    print(f"  Severity: {data['severity']}")
    print(f"  Hysteresis: {data['hysteresis_count']} consecutive breaches")

def threshold_list(
    active_only: bool = False,
    asset_id: str | None = None,
    format: str = "table",
    api_url: str = "http://localhost:8000",
    api_key: str | None = None,
) -> None:
    """List all thresholds.

    Shows threshold ID, state, metric condition, severity, and breach count.
    """
    params: dict[str, str] = {}
    if active_only:
        params["active_only"] = "true"
    if asset_id:
        params["asset_id"] = asset_id
    resp = httpx.get(
        f"{api_url}/api/thresholds",
        params=params,
        headers=_auth_headers(api_key),
    )
    resp.raise_for_status()
    thresholds = resp.json()
    if format == "json":
        print(json.dumps(thresholds, indent=2))
        return
    # Table output
    print(f"{'ID':<10} {'STATE':<12} {'METRIC':<20} {'CONDITION':<15} {'SEV':<10} {'BREACHES'}")
    print("-" * 80)
    for t in thresholds:
        condition = f"{t['operator']} {t['value']}"
        print(
            f"{t['id'][:8]:<10} "
            f"{t['state']:<12} "
            f"{t['metric_name'][:18]:<20} "
            f"{condition:<15} "
            f"{t['severity']:<10} "
            f"{t['consecutive_breaches']}"
        )
```

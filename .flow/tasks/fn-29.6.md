### Test Patterns
```python
from typer.testing import CliRunner
from unittest.mock import MagicMock, patch

runner = CliRunner(mix_stderr=False)

def test_run_start_json():
    mock_event = MagicMock()
    mock_event.event = "run_completed"
    mock_event.is_terminal = True
    mock_event.model_dump.return_value = {"event": "run_completed"}

    with patch("dataing_cli.commands.run.get_client") as mock_get:
        mock_client = MagicMock()
        mock_client.stream_run.return_value = [mock_event]
        mock_get.return_value = mock_client

        result = runner.invoke(app, ["run", "start", "test", "--goal", "test", "--json"])

        assert result.exit_code == 0
        assert '{"event":"run_completed"}' in result.output
```

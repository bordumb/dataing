**Example migration:**
```python
# Old test (uses tick)
async def test_workflow_tick_executes_step():
    workflow = Workflow()
    workflow.add_step(TestStep())
    result = await workflow.tick(context, "test_step")
    assert result.signal == Signal.CONTINUE

# New test (uses Runner directly or through run())
async def test_runner_executes_step():
    engine = Engine(step_order=["test_step"], merge_strategy=DefaultMergeStrategy())
    runner = Runner(engine=engine, steps={"test_step": TestStep()})
    outcome = await runner.run("run_1", context, "test_step")
    assert outcome.status == "completed"
```

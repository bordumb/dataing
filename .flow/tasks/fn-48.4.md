# fn-48.4 Implement threshold evaluation engine with state machine

## Description

Build the core evaluation engine that periodically checks active thresholds against current metric values and manages state transitions (normal -> breached -> recovering -> normal). The engine runs as a scheduled job (using the fn-47 scheduling infrastructure or a standalone Temporal cron) and evaluates each threshold by fetching the current metric value, comparing against the threshold condition, and transitioning state.

Depends on: fn-48.2 (repository and model must exist).

1. Create `python-packages/dataing/src/dataing/core/threshold/engine.py`:

```python
"""Threshold evaluation engine with state machine."""

from dataclasses import dataclass
from datetime import datetime, UTC
from decimal import Decimal
from enum import Enum


class ThresholdState(str, Enum):
    """Threshold state machine states."""
    NORMAL = "normal"
    BREACHED = "breached"
    RECOVERING = "recovering"


OPERATORS = {
    "gt": lambda measured, target: measured > target,
    "lt": lambda measured, target: measured < target,
    "eq": lambda measured, target: measured == target,
    "gte": lambda measured, target: measured >= target,
    "lte": lambda measured, target: measured <= target,
    "neq": lambda measured, target: measured != target,
}


@dataclass
class EvaluationResult:
    """Result of evaluating a single threshold."""
    threshold_id: str
    measured_value: Decimal
    is_breach: bool
    previous_state: ThresholdState
    new_state: ThresholdState
    should_trigger: bool  # True if investigation should be auto-triggered
    consecutive_breaches: int


class ThresholdEvaluationEngine:
    """Evaluates thresholds and manages state transitions.

    State machine:
      normal    + breach  -> breached (increment consecutive_breaches)
      normal    + ok      -> normal (no change)
      breached  + breach  -> breached (increment consecutive_breaches)
      breached  + ok      -> recovering
      recovering + ok     -> normal (reset consecutive_breaches)
      recovering + breach -> breached
    """

    def evaluate(
        self,
        threshold: Threshold,
        measured_value: Decimal,
    ) -> EvaluationResult:
        """Evaluate a threshold against a measured value."""
        op_fn = OPERATORS[threshold.operator]
        is_breach = op_fn(measured_value, threshold.value)
        prev_state = ThresholdState(threshold.state)

        # State transitions
        if is_breach:
            new_state = ThresholdState.BREACHED
            consecutive = threshold.consecutive_breaches + 1
        else:
            if prev_state == ThresholdState.BREACHED:
                new_state = ThresholdState.RECOVERING
                consecutive = threshold.consecutive_breaches
            elif prev_state == ThresholdState.RECOVERING:
                new_state = ThresholdState.NORMAL
                consecutive = 0
            else:
                new_state = ThresholdState.NORMAL
                consecutive = 0

        # Determine if we should trigger
        should_trigger = (
            is_breach
            and consecutive >= threshold.hysteresis_count
            and self._cooldown_elapsed(threshold)
        )

        return EvaluationResult(
            threshold_id=str(threshold.id),
            measured_value=measured_value,
            is_breach=is_breach,
            previous_state=prev_state,
            new_state=new_state,
            should_trigger=should_trigger,
            consecutive_breaches=consecutive,
        )
```

2. Create `python-packages/dataing/src/dataing/core/threshold/metric_collector.py`:

```python
class MetricCollector:
    """Collects metric values for threshold evaluation.

    Supports two metric sources:
    - "query": Execute a SQL query against the asset's datasource
    - "api": Fetch metric from external monitoring API (future)
    """

    async def collect(
        self,
        threshold: Threshold,
        datasource_adapter: Any,
    ) -> Decimal:
        """Collect the current metric value for a threshold."""
        if threshold.metric_source == "query":
            return await self._query_metric(threshold, datasource_adapter)
        raise ValueError(f"Unknown metric source: {threshold.metric_source}")
```

3. Create a Temporal activity `evaluate_thresholds` that:
   - Loads all active thresholds due for evaluation
   - For each threshold, collects the metric value and runs the evaluation engine
   - Updates threshold state in the database
   - Records evaluation in `threshold_evaluations` table
   - Returns list of thresholds that should trigger investigations

4. Create a Temporal workflow `ThresholdEvaluationWorkflow` (cron) that runs the evaluation activity on a schedule (default: every 5 minutes).

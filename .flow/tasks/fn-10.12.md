import pytest
import asyncio
from uuid import uuid4

from dataing.core.queue import enqueue_investigation
from dataing.entrypoints.worker import run_investigation
from dataing.adapters.db.app_db import AppDatabase


@pytest.fixture
async def db():
    db = await AppDatabase.create()
    yield db
    await db.close()


@pytest.fixture
async def redis_pool():
    from dataing.core.queue import get_queue
    pool = await get_queue()
    yield pool
    await pool.close()


class TestDurableExecution:
    @pytest.mark.asyncio
    async def test_investigation_queued_and_processed(self, db, redis_pool):
        # Create test investigation
        investigation_id = uuid4()
        tenant_id = uuid4()
        await create_test_investigation(db, investigation_id, tenant_id)

        # Enqueue
        job_id = await enqueue_investigation(str(investigation_id), str(tenant_id))
        assert job_id is not None

        # Verify job record created
        job = await db.fetchrow(
            "SELECT * FROM investigation_jobs WHERE investigation_id = $1",
            investigation_id
        )
        assert job["status"] == "pending"

    @pytest.mark.asyncio
    async def test_checkpoint_survives_restart(self, db):
        # Create job with existing checkpoint
        job_id = await create_job_with_checkpoint(db, step="generate_hypotheses")

        # Simulate worker restart by loading from checkpoint

            investigation_id=str(job_id),
            tenant_id="test"
        )

        assert result["resumed_from"] == "generate_hypotheses"

        # Verify checkpoint saved

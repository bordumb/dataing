import pytest
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import SimpleSpanProcessor
from opentelemetry.sdk.trace.export.in_memory_span_exporter import InMemorySpanExporter

@pytest.fixture
def isolated_tracer():
    """Create isolated tracer for test - doesn't affect global state."""
    exporter = InMemorySpanExporter()
    provider = TracerProvider()
    provider.add_span_processor(SimpleSpanProcessor(exporter))

    # Save original provider
    original_provider = trace.get_tracer_provider()

    # Set test provider
    trace.set_tracer_provider(provider)

    yield exporter

    # Restore original provider
    trace.set_tracer_provider(original_provider)
    exporter.clear()

@pytest.fixture
def reset_telemetry():
    """Reset telemetry state between tests."""
    from dataing.telemetry import config
    config._initialized = False
    yield
    config._initialized = False

@pytest.mark.asyncio
async def test_trace_propagation_api_to_worker(
    test_client: AsyncClient,

    # Create investigation via API

        json={"alert_summary": "Test anomaly"},
        headers={"X-Correlation-ID": "test-correlation-123"},
    )

    assert response.status_code == 201
    assert "X-Correlation-ID" in response.headers
    assert response.headers["X-Correlation-ID"] == "test-correlation-123"

    investigation_id = response.json()["id"]

    # Process job (in test, we trigger directly)
    await process_pending_jobs()

    # Verify spans
    spans = isolated_tracer.get_finished_spans()

    # Find key spans
    api_spans = [s for s in spans if "POST" in s.name or "investigations" in s.name]
    producer_spans = [s for s in spans if s.name == "investigation.enqueue"]
    consumer_spans = [s for s in spans if s.name == "investigation.process"]
    step_spans = [s for s in spans if s.name.startswith("workflow.step")]

    assert len(api_spans) >= 1, "Should have API span"
    assert len(producer_spans) >= 1, "Should have producer span"
    assert len(consumer_spans) >= 1, "Should have consumer span"

    # Verify same trace ID across all spans
    api_trace_id = api_spans[0].context.trace_id

    for span in producer_spans + consumer_spans:
        assert span.context.trace_id == api_trace_id, \
            f"Span {span.name} should be in same trace"

    # Verify correlation ID in consumer span
    consumer_span = consumer_spans[0]
    assert consumer_span.attributes.get("correlation_id") == "test-correlation-123"

@pytest.mark.asyncio
async def test_no_pii_in_spans(
    test_client: AsyncClient,
    isolated_tracer: InMemorySpanExporter,
):
    """Verify no sensitive data in span attributes."""
    response = await test_client.post(
        "/api/v1/investigations",
        json={"alert_summary": "Contains PII: user@email.com"},
    )

    await process_pending_jobs()

    spans = isolated_tracer.get_finished_spans()

    for span in spans:
        for key, value in span.attributes.items():
            assert "email" not in str(value).lower(), \
                f"PII found in span {span.name} attribute {key}"
            assert "input_data" not in key, \
                f"input_data attribute found in span {span.name}"

@pytest.mark.asyncio
async def test_correlation_id_in_structlog(
    test_client: AsyncClient,
    caplog,
):
    """Verify correlation ID appears in structured logs."""
    correlation_id = "log-test-456"

    with caplog.at_level("INFO"):
        response = await test_client.post(
            "/api/v1/investigations",
            json={"alert_summary": "Test"},
            headers={"X-Correlation-ID": correlation_id},
        )

        await process_pending_jobs()

    # Check for correlation ID in logs
    log_messages = [r.message for r in caplog.records]
    assert any(correlation_id in msg for msg in log_messages), \
        f"Correlation ID should appear in logs. Got: {log_messages[:5]}"

# fn-46.6 Add CLI commands: workspace list, workspace switch

## Description

Add CLI commands for workspace management so developers and operators can list available workspaces
and switch the active workspace context from the command line. The CLI stores the active workspace
in a local config file and sends it as a header on subsequent API calls. This is essential for
scripts and automation that need to target specific workspaces.

Depends on: fn-46.3 (workspace CRUD API routes must exist to call).

### Implementation

1. Create the CLI module `python-packages/dataing/src/dataing/entrypoints/cli/__init__.py` and
   `python-packages/dataing/src/dataing/entrypoints/cli/workspace.py`:
   ```python
   """Workspace CLI commands."""
   import click
   import httpx
   import json
   from pathlib import Path

   DATAING_CONFIG_DIR = Path.home() / ".dataing"
   DATAING_CONFIG_FILE = DATAING_CONFIG_DIR / "config.json"

   def _load_config() -> dict:
       """Load CLI config from ~/.dataing/config.json."""
       if DATAING_CONFIG_FILE.exists():
           return json.loads(DATAING_CONFIG_FILE.read_text())
       return {}

   def _save_config(config: dict) -> None:
       """Save CLI config to ~/.dataing/config.json."""
       DATAING_CONFIG_DIR.mkdir(parents=True, exist_ok=True)
       DATAING_CONFIG_FILE.write_text(json.dumps(config, indent=2))

   @click.group()
   def workspace():
       """Manage workspaces."""
       pass

   @workspace.command("list")
   @click.option("--api-url", envvar="DATAING_API_URL", default="http://localhost:8000")
   @click.option("--api-key", envvar="DATAING_API_KEY", required=True)
   def list_workspaces(api_url: str, api_key: str):
       """List all workspaces in the current tenant."""
       response = httpx.get(
           f"{api_url}/api/v1/workspaces",
           headers={"Authorization": f"Bearer {api_key}"},
       )
       response.raise_for_status()
       workspaces = response.json()

       config = _load_config()
       active_id = config.get("workspace_id")

       for ws in workspaces:
           marker = " *" if ws["id"] == active_id else ""
           default = " (default)" if ws.get("is_default") else ""
           click.echo(f"  {ws['slug']}{default}{marker}  {ws['name']}  [{ws['id']}]")

   @workspace.command("switch")
   @click.argument("slug_or_id")
   @click.option("--api-url", envvar="DATAING_API_URL", default="http://localhost:8000")
   @click.option("--api-key", envvar="DATAING_API_KEY", required=True)
   def switch_workspace(slug_or_id: str, api_url: str, api_key: str):
       """Switch the active workspace."""
       # Resolve slug to ID via API
       response = httpx.get(
           f"{api_url}/api/v1/workspaces",
           headers={"Authorization": f"Bearer {api_key}"},
       )
       response.raise_for_status()
       workspaces = response.json()

       match = None
       for ws in workspaces:
           if ws["slug"] == slug_or_id or ws["id"] == slug_or_id:
               match = ws
               break

       if not match:
           click.echo(f"Workspace not found: {slug_or_id}", err=True)
           raise SystemExit(1)

       config = _load_config()
       config["workspace_id"] = match["id"]
       config["workspace_slug"] = match["slug"]
       _save_config(config)
       click.echo(f"Switched to workspace: {match['name']} ({match['slug']})")

   @workspace.command("current")
   def current_workspace():
       """Show the currently active workspace."""
       config = _load_config()
       ws_id = config.get("workspace_id")
       ws_slug = config.get("workspace_slug")
       if ws_id:
           click.echo(f"Active workspace: {ws_slug} [{ws_id}]")
       else:
           click.echo("No workspace selected (using default)")
   ```

2. Add `click` to dependencies if not already present in
   `python-packages/dataing/pyproject.toml`.

3. Register the CLI entrypoint in `pyproject.toml`:
   ```toml
   [project.scripts]
   dataing = "dataing.entrypoints.cli:main"
   ```

4. Create the main CLI group in `__init__.py`:
   ```python
   import click
   from dataing.entrypoints.cli.workspace import workspace

   @click.group()
   def main():
       """Dataing CLI."""
       pass

   main.add_command(workspace)
   ```

   - Test `workspace switch` saves config.
   - Test `workspace current` reads config.
   - Test workspace not found error.
   - Use `click.testing.CliRunner` and mock httpx responses.

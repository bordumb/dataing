@pytest.mark.asyncio
async def test_replay_produces_identical_context():
    """Core invariant: replay(events) == live_run.context"""
    # Setup
    workflow = create_test_workflow()
    engine = Engine(step_order=workflow.step_order, merge_strategy=DefaultMergeStrategy())
    runner = Runner(engine=engine, steps=workflow.steps)
    replayer = Replayer(engine=engine)

    # Live run
    initial_context = {"count": 0}
    outcome = await runner.run("test_run", initial_context, "step_a")

    # Replay
    initial_state = RunState(
        run_id="test_run",
        status="running",
        context=initial_context,
        current_step="step_a",
        seq=0
    )
    replayed_state = replayer.replay(initial_state, outcome.events)

    # Assert identical
    assert replayed_state.context == outcome.context
    assert replayed_state.status == "completed"

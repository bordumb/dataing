```python
def get_current_investigation_id(self) -> str | None:
    """Get the current investigation ID from config."""
    return self._config.get("session", {}).get("current_investigation_id")

def set_current_investigation_id(self, investigation_id: str | None) -> None:
    """Set the current investigation ID in config."""
    if "session" not in self._config:
        self._config["session"] = {}
    self._config["session"]["current_investigation_id"] = investigation_id
    self._save_config()
```

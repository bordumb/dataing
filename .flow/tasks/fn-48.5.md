# fn-48.5 Implement auto-trigger investigation on breach with cooldown

## Description

When the threshold evaluation engine detects a breach that passes hysteresis requirements, automatically start an investigation. The investigation should include the threshold context (what metric breached, the measured value, the threshold value) in its alert data so the LLM has full context. A cooldown mechanism prevents the same threshold from triggering multiple investigations within a configurable window.

Depends on: fn-48.4 (evaluation engine must exist to produce trigger signals).

### Implementation

1. Create `python-packages/dataing/src/dataing/core/threshold/trigger.py`:

```python
"""Auto-trigger investigation on threshold breach."""

from datetime import datetime, UTC, timedelta
from uuid import uuid4

from dataing.core.threshold.engine import EvaluationResult
from dataing.temporal.client import TemporalInvestigationClient


class ThresholdInvestigationTrigger:
    """Triggers investigations when thresholds are breached."""

    def __init__(
        self,
        temporal_client: TemporalInvestigationClient,
        threshold_repo: ThresholdRepository,
    ) -> None:
        """Initialize the trigger."""
        self._temporal = temporal_client
        self._repo = threshold_repo

    async def maybe_trigger(
        self,
        threshold: Threshold,
        eval_result: EvaluationResult,
    ) -> str | None:
        """Trigger an investigation if the evaluation warrants it.

        Returns the investigation_id if triggered, None otherwise.
        """
        if not eval_result.should_trigger:
            return None

        # Build alert data with threshold context
        alert_data = {
            "anomaly_type": "threshold_breach",
            "dataset_ids": [str(threshold.asset_id)],
            "metric_spec": {
                "metric_type": "threshold",
                "expression": threshold.metric_query or "",
                "display_name": threshold.metric_name,
                "columns_referenced": [],
            },
            "expected_value": float(threshold.value),
            "actual_value": float(eval_result.measured_value),
            "deviation_pct": self._calc_deviation(
                threshold.value, eval_result.measured_value
            ),
            "anomaly_date": datetime.now(UTC).isoformat(),
            "severity": threshold.severity,
            "metadata": {
                "threshold_id": str(threshold.id),
                "operator": threshold.operator,
                "consecutive_breaches": eval_result.consecutive_breaches,
                "state_transition": f"{eval_result.previous_state} -> {eval_result.new_state}",
            },
        }

        investigation_id = str(uuid4())
        alert_summary = (
            f"Threshold breach: {threshold.metric_name} "
            f"{threshold.operator} {threshold.value} "
            f"(measured: {eval_result.measured_value})"
        )

        await self._temporal.start_investigation(
            investigation_id=investigation_id,
            tenant_id=str(threshold.tenant_id),
            datasource_id="",  # Resolved from asset
            alert_data=alert_data,
            alert_summary=alert_summary,
        )

        # Update last_triggered_at for cooldown tracking
        await self._repo.update_state(
            threshold_id=threshold.id,
            state=eval_result.new_state,
            consecutive_breaches=eval_result.consecutive_breaches,
            last_value=eval_result.measured_value,
            last_triggered_at=datetime.now(UTC),
        )

        return investigation_id
```

2. Add cooldown check to `ThresholdEvaluationEngine._cooldown_elapsed()`:

```python
def _cooldown_elapsed(self, threshold: Threshold) -> bool:
    """Check if cooldown period has elapsed since last trigger."""
    if threshold.last_triggered_at is None:
        return True
    cooldown_end = threshold.last_triggered_at + timedelta(
        hours=threshold.cooldown_hours
    )
    return datetime.now(UTC) >= cooldown_end
```

3. Wire `ThresholdInvestigationTrigger` into the evaluation Temporal activity so that after evaluating all thresholds, those with `should_trigger=True` automatically start investigations.

4. Record the `investigation_id` in the `threshold_evaluations` row for cross-referencing.

5. Add deduplication: before triggering, check if an investigation for the same threshold+asset is already running (via Temporal workflow status query). Skip if running.

## Implementation

### ask.py

```python
import typer
from typing import Optional

from dataing_cli.config import get_config
from dataing_cli.errors import cli_error_handler

app = typer.Typer(name="ask", help="Interactive investigation mode")

@app.callback(invoke_without_command=True)
@cli_error_handler
def ask(
    ctx: typer.Context,
    question: Optional[str] = typer.Argument(None, help="Question to ask (one-shot mode)"),
    investigation: Optional[str] = typer.Option(
        None, "--investigation", "-i", help="Investigation ID to attach to"
    ),
) -> None:
    """Ask questions about an investigation.

    With no args: enters interactive REPL mode
    With question: sends one-shot question and streams response
    """
    config = get_config()
    state = ctx.obj

    # Resolve investigation ID
    inv_id = investigation or config.get_current_investigation_id()
    if not inv_id:
        raise typer.BadParameter(
            "No investigation attached. Use --investigation <id> or run 'dataing run start' first."
        )

    # Save as current investigation if explicitly provided
    if investigation:
        config.set_current_investigation_id(investigation)

    if question:
        # One-shot mode
        _send_one_shot(state, inv_id, question)
    else:
        # Interactive mode (implemented in fn-31.4)
        _enter_repl(state, inv_id)


def _send_one_shot(state, investigation_id: str, question: str) -> None:
    """Send a single question and stream the response."""
    from rich.console import Console
    from rich.live import Live

    console = Console()
    client = state.client

    # Send message
    client.send_message(investigation_id, question)

    # Stream response events
    with Live(console=console, transient=False) as live:
        for event in client.stream_run(investigation_id):
            # Display streaming events using existing display helpers
            from dataing_cli.display import format_event
            live.update(format_event(event))
            if event.is_terminal:
                break


def _enter_repl(state, investigation_id: str) -> None:
    """Enter interactive REPL mode."""
    # Placeholder - implemented in fn-31.4
    from dataing_cli.repl import DataingREPL
    repl = DataingREPL(state, investigation_id)
    repl.run()
```

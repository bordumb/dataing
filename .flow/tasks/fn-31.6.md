import pytest
from typer.testing import CliRunner
from unittest.mock import Mock, patch

from dataing_cli.main import app


class TestAskCommand:
    """Tests for dataing ask command."""

    def test_ask_help(self, runner):
        """Test ask --help shows usage."""
        result = runner.invoke(app, ["ask", "--help"])
        assert result.exit_code == 0
        assert "Interactive investigation mode" in result.output

    def test_ask_no_investigation_error(self, runner, mock_config_no_investigation):
        """Test error when no investigation attached."""
        result = runner.invoke(app, ["ask", "test question"])
        assert result.exit_code != 0
        assert "No investigation attached" in result.output

    def test_ask_one_shot_sends_message(self, runner, mock_client):
        """Test one-shot mode sends message via SDK."""
        with patch("dataing_cli.commands.ask.get_config") as mock_cfg:
            mock_cfg.return_value.get_current_investigation_id.return_value = "test-id"
            result = runner.invoke(app, ["ask", "What caused this?"])

        mock_client.send_message.assert_called_once_with("test-id", "What caused this?")

    def test_ask_with_investigation_flag(self, runner, mock_client):
        """Test --investigation flag overrides config."""
        result = runner.invoke(app, ["ask", "--investigation", "explicit-id", "question"])

        mock_client.send_message.assert_called_with("explicit-id", "question")

    def test_ask_saves_investigation_to_config(self, runner, mock_config):
        """Test explicit investigation ID saved to config."""
        runner.invoke(app, ["ask", "--investigation", "new-id", "question"])

        mock_config.set_current_investigation_id.assert_called_with("new-id")


class TestAskOneShot:
    """Tests for one-shot message sending."""

    def test_streams_response(self, runner, mock_client):
        """Test response is streamed to terminal."""
        # Mock stream_run to return events
        mock_client.stream_run.return_value = [
            Mock(is_terminal=False, type="thinking"),
            Mock(is_terminal=True, type="synthesis"),
        ]

        result = runner.invoke(app, ["ask", "question"])

        mock_client.stream_run.assert_called_once()

    def test_stops_on_terminal_event(self, runner, mock_client):
        """Test streaming stops on terminal event."""
        events = [Mock(is_terminal=True)]
        mock_client.stream_run.return_value = iter(events)

        runner.invoke(app, ["ask", "question"])
        # Should not hang or error

import pytest
from unittest.mock import Mock, AsyncMock, patch

from dataing_cli.repl import DataingREPL


class TestDataingREPL:
    """Tests for REPL functionality."""

    @pytest.fixture
    def mock_state(self):
        state = Mock()
        state.client = Mock()
        return state

    @pytest.fixture
    def repl(self, mock_state):
        with patch("dataing_cli.repl.FileHistory"):
            return DataingREPL(mock_state, "test-investigation-id")

    def test_build_completer(self, repl):
        """Test completer includes all slash commands."""
        completer = repl._build_completer()
        # NestedCompleter should have our commands
        assert "/help" in str(completer)

    def test_get_prompt_format(self, repl):
        """Test prompt shows investigation context."""
        repl.investigation = Mock(hypotheses=[1, 2], evidence=[1, 2, 3])
        prompt = repl._get_prompt()
        assert "test-inv" in str(prompt)

    @pytest.mark.asyncio
    async def test_handle_command_help(self, repl):
        """Test /help command."""
        await repl._cmd_help([])
        # Should not raise

    @pytest.mark.asyncio
    async def test_handle_command_quit(self, repl):
        """Test /quit raises EOFError."""
        with pytest.raises(EOFError):
            await repl._cmd_quit([])

    @pytest.mark.asyncio
    async def test_handle_message_sends_to_sdk(self, repl):
        """Test message is sent via SDK."""
        repl.state.client.stream_run.return_value = [Mock(is_terminal=True)]

        await repl._handle_message("test question")

        repl.state.client.send_message.assert_called_once()


class TestSlashCommands:
    """Tests for individual slash commands."""

    @pytest.mark.asyncio
    async def test_hypotheses_empty(self, repl):
        """Test /hypotheses with no hypotheses."""
        repl.investigation = Mock(hypotheses=[])
        await repl._cmd_hypotheses([])
        # Should print "No hypotheses"

    @pytest.mark.asyncio
    async def test_evidence_empty(self, repl):
        """Test /evidence with no evidence."""
        repl.investigation = Mock(evidence=[])
        await repl._cmd_evidence([])
        # Should print "No evidence"

    @pytest.mark.asyncio
    async def test_export_default_markdown(self, repl):
        """Test /export defaults to markdown."""
        with patch("dataing_cli.repl.export_investigation") as mock_export:
            await repl._cmd_export([])
            mock_export.assert_called_with(repl.investigation_id, format="markdown")

## Description

Create the Temporal worker entrypoint with full dependency injection.

### Implementation

```python
# dataing/src/dataing/entrypoints/temporal_worker.py

async def run_worker():
    client = await Client.connect(
        target_host=settings.TEMPORAL_HOST,
        namespace=settings.TEMPORAL_NAMESPACE,
    )

    # Create dependencies
    context_engine = create_context_engine()
    database = create_database_adapter()
    agent_client = create_agent_client()

    # Wire activities via factories
    activities = [
        make_gather_context_activity(context_engine),
        make_check_patterns_activity(context_engine),
        make_generate_hypotheses_activity(agent_client),
        # ... all 8 activities
    ]

    worker = Worker(
        client,
        task_queue=settings.TEMPORAL_TASK_QUEUE,
        workflows=[InvestigationWorkflow, EvaluateHypothesisWorkflow],
        activities=activities,
        max_concurrent_activities=10,
        max_concurrent_workflow_tasks=5,
    )

    await worker.run()
```

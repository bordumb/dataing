import pytest
from unittest.mock import AsyncMock, patch

from dataing.core.queue import get_redis_settings, enqueue_investigation


class TestQueueSettings:
    def test_redis_settings_from_env(self, monkeypatch):
        monkeypatch.setenv("REDIS_HOST", "test-redis")
        monkeypatch.setenv("REDIS_PORT", "6380")

        settings = get_redis_settings()

        assert settings.host == "test-redis"
        assert settings.port == 6380

    def test_redis_settings_defaults(self):
        settings = get_redis_settings()

        assert settings.host == "localhost"
        assert settings.port == 6379


class TestEnqueueInvestigation:
    @pytest.mark.asyncio
    async def test_enqueue_calls_arq(self):
        with patch("dataing.core.queue.get_queue") as mock_pool:
            mock_pool.return_value.enqueue_job = AsyncMock(return_value=MagicMock(job_id="test-123"))

            job_id = await enqueue_investigation("inv-1", "tenant-1")

            mock_pool.return_value.enqueue_job.assert_called_once_with(
                "run_investigation",
                investigation_id="inv-1",
                tenant_id="tenant-1",
                _queue_name="investigations",
            )

import pytest
import asyncio
import signal
from unittest.mock import AsyncMock, MagicMock, patch

from dataing.entrypoints.worker import WorkerSettings, startup, shutdown


class TestWorkerSettings:
    def test_worker_settings_configured(self):
        assert WorkerSettings.max_jobs == 5
        assert WorkerSettings.job_timeout == 7200
        assert WorkerSettings.job_completion_wait == 30
        assert WorkerSettings.retry_jobs is False


class TestWorkerLifecycle:
    @pytest.mark.asyncio
    async def test_startup_creates_db(self):
        ctx = {}
        with patch("dataing.entrypoints.worker.AppDatabase") as mock_db:
            mock_db.create = AsyncMock()
            await startup(ctx)
            assert "db" in ctx

    @pytest.mark.asyncio
    async def test_shutdown_closes_db(self):
        mock_db = AsyncMock()
        ctx = {"db": mock_db}
        await shutdown(ctx)
        mock_db.close.assert_called_once()

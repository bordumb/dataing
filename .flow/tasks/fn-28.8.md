# fn-28.8 Rich output helpers: format_event, format_table, format_status

## Description


Create Rich output helpers for formatting events, tables, and status displays.

### File: `src/dataing_cli/output.py`

```python
"""Rich output formatting helpers."""
from __future__ import annotations

from typing import Any

from rich.console import Console
from rich.panel import Panel
from rich.syntax import Syntax
from rich.table import Table

console = Console()


def format_event(event: Any) -> Panel:
    """Format a streaming event for display."""
    event_type = event.type

    if event_type == "evidence":
        hypothesis = event.hypothesis or "Unknown"
        query = event.query or ""
        interpretation = event.interpretation or ""
        supports = event.supports_hypothesis

        icon = "✓" if supports else "✗" if supports is False else "?"
        color = "green" if supports else "red" if supports is False else "yellow"

        content = f"[bold]{hypothesis}[/bold]\n\n"

        if query:
            content += "[dim]Query:[/dim]\n"
            syntax = Syntax(query, "sql", theme="monokai", line_numbers=False)
            content += str(syntax) + "\n\n"

        if interpretation:
            content += f"[dim]Finding:[/dim] {interpretation}"

        return Panel(
            content,
            title=f"[{color}]{icon}[/{color}] Evidence",
            border_style=color,
        )

    # Default: just show raw data
    return Panel(str(event), title=event_type)


def format_completion(event: Any, console: Console) -> None:
    """Print completion summary."""
    root_cause = event.root_cause or "No root cause identified"
    confidence = event.confidence or 0

    console.print(Panel(
        f"[green]✓[/green] Investigation complete\n\n"
        f"[bold]Root Cause:[/bold] {root_cause}\n"
        f"[bold]Confidence:[/bold] {confidence:.0%}",
        title="Result",
        border_style="green",
    ))


def print_status(config: dict, api_key: str | None, connected: bool, error: str | None = None) -> None:
    """Print connection status panel."""
    table = Table(show_header=False, box=None)
    table.add_column("Key", style="dim")
    table.add_column("Value")

    table.add_row("Backend", config.get("api_url", "Not configured"))
    table.add_row("API Key", f"***{api_key[-4:]}" if api_key else "Not configured")
    table.add_row("Default DS", config.get("default_datasource_name", "None"))

    status = "[green]● Connected[/green]" if connected else "[red]● Disconnected[/red]"
    if error:
        status += f"\n  [dim]{error}[/dim]"
    table.add_row("Status", status)

    console.print(Panel(table, title="dataing CLI"))


def print_datasources_table(datasources: list, console: Console) -> None:
    """Print datasources in table format."""
    table = Table(title="Datasources")
    table.add_column("ID", style="cyan")
    table.add_column("Name")
    table.add_column("Type")
    table.add_column("Status")

    for ds in datasources:
        status = "[green]●[/green]" if ds.status == "connected" else "[red]●[/red]"
        table.add_row(ds.id[:12] + "...", ds.name, ds.source_type, status)

    console.print(table)
```

### Key Patterns

- Use Rich markup for colors: `[green]`, `[red]`, `[bold]`
- Use `Panel` for boxed content with titles
- Use `Table` for structured data
- Use `Syntax` for SQL highlighting
- Escape user input with `rich.markup.escape()` if displaying untrusted content

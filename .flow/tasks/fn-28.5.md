## Description


Implement `dataing init` and `dataing status` commands.

### `init` Command

```python
@app.command()
def init(
    ctx: typer.Context,
    url: Annotated[str, typer.Option("--url", "-u", prompt="Backend URL")] = "http://localhost:8000",
    api_key: Annotated[str, typer.Option("--api-key", "-k", prompt=True, hide_input=True)] = ...,
    no_keyring: Annotated[bool, typer.Option("--no-keyring", help="Store in config file instead of keyring")] = False,
):
    """Initialize CLI configuration."""
    from dataing_cli.config import save_config, save_api_key, get_config_path
    from dataing_sdk import DataingClient

    # Test connection first
    try:
        client = DataingClient(base_url=url, api_key=api_key)
        client.health()
    except Exception as e:
        console.print(f"[red]Connection failed:[/red] {e}")
        raise typer.Exit(1)

    # Save config
    save_config({"api_url": url})
    save_api_key(api_key, use_keyring=not no_keyring)

    if no_keyring:
        console.print("[yellow]Warning:[/yellow] API key stored in config file. Consider using keyring.")

    console.print(f"[green]✓[/green] Configuration saved to {get_config_path()}")
```

### `status` Command

```python
@app.command()
def status(ctx: typer.Context):
    """Check connection and show current configuration."""
    from dataing_cli.config import load_config, get_api_key, get_client
    from dataing_cli.output import print_status

    state = ctx.obj
    config = load_config()
    api_key = get_api_key(state.api_key)

    try:
        client = get_client(state.api_key, state.base_url)
        client.health()
        connected = True
        error = None
    except Exception as e:
        connected = False
        error = str(e)

    if state.json_output:
        # Output JSON
        console.print_json({"connected": connected, "url": config.get("api_url"), ...})
    else:
        print_status(config, api_key, connected, error)

    if not connected:
        raise typer.Exit(1)
```

### Non-Interactive Mode

For CI/CD, `init` should work without prompts if all flags are provided:

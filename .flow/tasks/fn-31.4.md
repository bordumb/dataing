```python
import asyncio
from pathlib import Path

from prompt_toolkit import PromptSession
from prompt_toolkit.completion import NestedCompleter
from prompt_toolkit.history import FileHistory, ThreadedHistory
from prompt_toolkit.patch_stdout import patch_stdout
from prompt_toolkit.formatted_text import HTML
from rich.console import Console
from rich.live import Live

from dataing_cli.config import get_history_path
from dataing_cli.display import format_event, format_hypothesis, format_evidence_item


class DataingREPL:
    """Interactive REPL for investigation conversations."""

    def __init__(self, state, investigation_id: str):
        self.state = state
        self.investigation_id = investigation_id
        self.console = Console()
        self.investigation = None  # Cached investigation state

        history_path = get_history_path()

        self.session = PromptSession(
            history=ThreadedHistory(FileHistory(str(history_path))),
            completer=self._build_completer(),
        )

    def _build_completer(self) -> NestedCompleter:
        """Build tab completer for slash commands."""
        return NestedCompleter.from_nested_dict({
            "/lineage": None,
            "/hypotheses": None,
            "/evidence": None,
            "/export": {"json": None, "markdown": None},
            "/help": None,
            "/quit": None,
        })

    def _get_prompt(self) -> HTML:
        """Build prompt with context indicator."""
        short_id = self.investigation_id[:8]
        hyp_count = len(self.investigation.hypotheses) if self.investigation else 0
        ev_count = len(self.investigation.evidence) if self.investigation else 0
        return HTML(
            f'<inv>[{short_id}]</inv> '
            f'<dim>hyp:{hyp_count} ev:{ev_count}</dim> > '
        )

    def run(self) -> None:
        """Run the REPL loop."""
        asyncio.run(self._run_async())

    async def _run_async(self) -> None:
        """Async REPL loop."""
        # Initial state refresh
        await self._refresh_investigation()

        self.console.print(f"[bold]Attached to investigation {self.investigation_id}[/bold]")
        self.console.print("[dim]Type /help for commands, Ctrl+C to cancel, Ctrl+D to exit[/dim]\n")

        with patch_stdout():
            while True:
                try:
                    text = await self.session.prompt_async(self._get_prompt)
                    await self._handle_input(text.strip())
                except KeyboardInterrupt:
                    self.console.print("\n[dim]Interrupted. Type /quit to exit.[/dim]")
                    continue
                except EOFError:
                    break

        self.console.print("[dim]Goodbye![/dim]")

    async def _handle_input(self, text: str) -> None:
        """Route input to handler."""
        if not text:
            return

        if text.startswith("/"):
            await self._handle_command(text)
        else:
            await self._handle_message(text)

    async def _handle_command(self, text: str) -> None:
        """Handle slash command."""
        parts = text.split()
        cmd = parts[0].lower()
        args = parts[1:]

        handlers = {
            "/lineage": self._cmd_lineage,
            "/hypotheses": self._cmd_hypotheses,
            "/evidence": self._cmd_evidence,
            "/export": self._cmd_export,
            "/help": self._cmd_help,
            "/quit": self._cmd_quit,
        }

        handler = handlers.get(cmd)
        if handler:
            await handler(args)
        else:
            self.console.print(f"[red]Unknown command: {cmd}[/red]")

    async def _handle_message(self, message: str) -> None:
        """Send message and stream response."""
        client = self.state.client

        # Send message
        client.send_message(self.investigation_id, message)

        # Stream response
        with Live(console=self.console, transient=False) as live:
            for event in client.stream_run(self.investigation_id):
                live.update(format_event(event))
                if event.is_terminal:
                    break

        # Refresh state after response
        await self._refresh_investigation()

    async def _refresh_investigation(self) -> None:
        """Refresh cached investigation state."""
        self.investigation = self.state.client.get_investigation(self.investigation_id)

    # Slash command implementations (in fn-31.5)
    async def _cmd_lineage(self, args): pass
    async def _cmd_hypotheses(self, args): pass
    async def _cmd_evidence(self, args): pass
    async def _cmd_export(self, args): pass
    async def _cmd_help(self, args): pass
    async def _cmd_quit(self, args): raise EOFError()
```

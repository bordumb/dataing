```python
"""Investigation run commands."""
from __future__ import annotations

from typing import Annotated, Optional

import typer
from rich.console import Console
from rich.live import Live
from rich.panel import Panel

from dataing_cli.output import format_event, format_completion

app = typer.Typer()
console = Console()


@app.command("start")
def start_run(
    ctx: typer.Context,
    dataset: Annotated[str, typer.Argument(help="Dataset to investigate (e.g., 'schema.table')")],
    goal: Annotated[str, typer.Option("--goal", "-g", help="Investigation goal")],
    datasource: Annotated[Optional[str], typer.Option("--datasource", "-d", help="Datasource ID")] = None,
    no_watch: Annotated[bool, typer.Option("--no-watch", help="Don't stream progress")] = False,
):
    """Start a new investigation run."""
    from dataing_cli.config import get_client, load_config
    from dataing_sdk import AssetRef

    state = ctx.obj
    config = load_config()

    ds_id = datasource or config.get("default_datasource_id")
    if not ds_id:
        console.print("[red]Error:[/red] No datasource specified. Use --datasource or run 'dataing ds attach'")
        raise typer.Exit(1)

    client = get_client(state.api_key, state.base_url)

    # Parse dataset into AssetRef
    asset = AssetRef(platform="unknown", name=dataset, datasource_id=ds_id)

    with console.status("[bold blue]Starting investigation..."):
        run = client.run(assets=[asset], goal=goal)

    console.print(f"[green]✓[/green] Started run: [cyan]{run.run_id}[/cyan]")

    if not no_watch:
        _watch_run(client, run.run_id, state)


@app.command("watch")
def watch_run(
    ctx: typer.Context,
    run_id: Annotated[str, typer.Argument(help="Run ID to watch")],
):
    """Watch a running investigation."""
    from dataing_cli.config import get_client

    state = ctx.obj
    client = get_client(state.api_key, state.base_url)
    _watch_run(client, run_id, state)


def _watch_run(client, run_id: str, state) -> None:
    """Stream and display run events with Rich Live."""
    console.print(f"\n[dim]Streaming events for run {run_id}...[/dim]\n")

    if state.json_output:
        # JSON mode: print events line by line
        for event in client.stream_run(run_id):
            console.print_json(event.model_dump())
        return

    with Live(Panel("Initializing...", title="Progress"), refresh_per_second=4, transient=True) as live:
        for event in client.stream_run(run_id):
            event_type = event.type

            if event_type == "progress":
                live.update(Panel(f"[yellow]⏳[/yellow] {event.message}", title="Progress"))

            elif event_type == "evidence":
                # Print evidence outside Live for persistence
                live.stop()
                console.print(format_event(event))
                live.start()

            elif event_type == "completed":
                live.stop()
                format_completion(event, console)
                return

            elif event_type == "failed":
                live.stop()
                console.print(Panel(
                    f"[red]✗[/red] {event.error or 'Unknown error'}",
                    title="Failed",
                    border_style="red",
                ))
                raise typer.Exit(1)
```

### Streaming Pattern

Use `Rich.Live` with `transient=True` for progress updates that disappear, but print evidence panels permanently so users can scroll back to see them.

Handle Ctrl+C gracefully - the generator from `stream_run()` will be interrupted, which is fine.

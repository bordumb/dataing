## Objective

Instrument worker to extract trace context and create properly linked spans.

## Critical Fix from Review

Setting `trace_id` as a span attribute does **not** create a parent-child relationship.
Must use `propagate.extract()` to restore the SpanContext and create linked spans.

## Deliverables

1. **Worker updates**: Proper context extraction and span linking
2. **Flow updates**: Child spans for steps (no PII)
3. **SLO metrics**: Using `enqueued_at` for E2E timing

### worker.py - Proper context restoration

```python
from datetime import datetime, timezone
from opentelemetry import trace
from opentelemetry.trace import SpanKind, Link
from opentelemetry.propagate import extract
from dataing.telemetry.config import init_telemetry, get_tracer
from dataing.telemetry.metrics import (
    record_investigation_duration,
    record_queue_wait_time,
)

# Initialize at worker startup
init_telemetry()

async def run_investigation(
    ctx: dict[str, Any],
    investigation_id: str,
    tenant_id: str,
    datasource_id: str | None,
    parent_job_id: str | None,
    branch_spec: dict[str, Any] | None,
    trace_context: dict[str, Any] | None,  # NEW: Contains traceparent, etc.
) -> dict[str, Any]:
    """Process investigation with proper trace context restoration."""

    tracer = get_tracer("dataing.worker")

    # Extract parent context from queue payload
    parent_ctx = None
    enqueued_at = None
    correlation_id = None

    if trace_context:
        # Restore W3C trace context (creates proper parent link)
        parent_ctx = extract(trace_context)
        enqueued_at = trace_context.get("enqueued_at")
        correlation_id = trace_context.get("correlation_id")

    # Calculate queue wait time for SLOs
    worker_start = datetime.now(timezone.utc)
    if enqueued_at:
        enqueued_dt = datetime.fromisoformat(enqueued_at)
        queue_wait_seconds = (worker_start - enqueued_dt).total_seconds()
        record_queue_wait_time(queue_wait_seconds)

    # Create consumer span linked to producer span
    with tracer.start_as_current_span(
        "investigation.process",
        context=parent_ctx,  # CRITICAL: Links to parent span
        kind=SpanKind.CONSUMER,
        attributes={
            # Safe attributes only - no PII
            "investigation.id": investigation_id,
            "tenant.id": tenant_id,
            "worker.id": WORKER_ID,
            "job.is_branch": parent_job_id is not None,
            "correlation_id": correlation_id or "",
            "messaging.system": "redis",
            "messaging.operation": "process",
        },
    ) as job_span:
        try:
            result = await _execute_workflow(...)

            job_span.set_attribute("investigation.status", "completed")

            # Record E2E duration (from enqueue to complete)
            if enqueued_at:
                e2e_duration = (datetime.now(timezone.utc) - enqueued_dt).total_seconds()
                record_investigation_duration(e2e_duration, "completed")

            return result

        except Exception as e:
            job_span.record_exception(e)
            job_span.set_attribute("investigation.status", "failed")

            if enqueued_at:
                e2e_duration = (datetime.now(timezone.utc) - enqueued_dt).total_seconds()
                record_investigation_duration(e2e_duration, "failed")

            raise
```
